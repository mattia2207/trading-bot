# 2. LOGICA DI CALCOLO METRICHE PROFITTEVOLE
def calcola_metriche(trades):
    trade_chiusi = [t for t in trades if t['status'] == 'CLOSED']
    totale_trade = len(trade_chiusi)
    
    if totale_trade == 0:
        return {"win_rate": 0, "profit_factor": 0, "total_pnl": 0, "total_trades": 0}
        
    vincenti = [t for t in trade_chiusi if t['pnl'] > 0]
    `perdenti` = [t for t in trade_chiusi if t['pnl'] <= 0]
    
    # Calcolo Win Rate
    win_rate = (len(vincenti) / totale_trade) * 100
    
    # Calcolo Profit Factor
    profitto_lordo = sum([t['pnl'] for t in vincenti])
    perdita_lorda = abs(sum([t['pnl'] for t in perdenti]))
    profit_factor = profitto_lordo / perdita_lorda if perdita_lorda > 0 else profitto_lordo
    
    # PnL Totale di tutto il bot (aperti + chiusi)
    total_pnl = sum([t['pnl'] for t in trades])
    
    return {
        "win_rate": round(win_rate, 1),
        "profit_factor": round(profit_factor, 2),
        "total_pnl": round(total_pnl, 2),
        "total_trades": totale_trade
    }
